## 2. Chạy trọn ca, không điều khiển (10800 tick = 180 s)
$ python3 -m chonoi.headless --mode gather --seed 7 --ticks 10800 --out gather.jsonl
{"type":"end","ticks":10800,"sim_tick":10800,"mode":"gather","state":"finished","reason":"time","money":0,"earned":0,"reputation":3.0,"score":300}
$ python3 -m chonoi.headless --mode quan --seed 7 --scenario q.json --ticks 10800 --out quan.jsonl   # q.json: {"quan":{"stock":{"ca_loc":2,"bong_sung":4,"ca_linh":3,"bong_dien_dien":1}}}
sự kiện quán:

{"type":"end","ticks":10800,"sim_tick":10800,"mode":"quan","state":"finished","reason":"time","money":0,"earned":0,"reputation":2.0,"score":200}
